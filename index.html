export default async function handler(req, res) { if (req.method !== "POST") { return res.status(405).json({ error: "Method not allowed" }); } try { const { origin, distance, bike, preferences } = req.body; if ( !origin || typeof origin.lat !== "number" || typeof origin.lng !== "number" ) { return res.status(400).json({ error: "Ubicación inválida" }); } /* * Aquí conectaremos el motor de routing. * * Ejemplo conceptual: * * const response = await fetch( * GRAPHOPPER_URL, * ... * ); * * La API key permanece en: * * process.env.ROUTING_API_KEY */ /* * IMPORTANTE: * * Esta respuesta es solamente un contrato * para que el frontend ya esté preparado. * * El siguiente paso es conectar aquí * GraphHopper/autohospedado. */ return res.status(501).json({ error: "Motor de rutas todavía no conectado", received: { origin, distance, bike, preferences } }); } catch (error) { console.error(error); return res.status(500).json({ error:"Error interno de routing" }); } }
